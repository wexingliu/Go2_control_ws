cmake_minimum_required(VERSION 3.16)
project(unitree_guide)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(POLICY CMP0144)
    cmake_policy(SET CMP0144 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Threads REQUIRED)
set(UNITREE_SDK2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../thirdparty/unitree_sdk2" CACHE PATH "Unitree SDK2 installation prefix")
find_package(unitree_sdk2 REQUIRED CONFIG PATHS "${UNITREE_SDK2_ROOT}/lib/cmake/unitree_sdk2" NO_DEFAULT_PATH)

file(GLOB_RECURSE CONTROLLER_SOURCES CONFIGURE_DEPENDS
    src/FSM/*.cpp src/Gait/*.cpp src/common/*.cpp src/control/*.cpp src/quadProgpp/*.cc)
add_executable(junior_ctrl src/main.cpp src/interface/KeyBoard.cpp src/interface/IODDS.cpp ${CONTROLLER_SOURCES})
target_include_directories(junior_ctrl PRIVATE include)
target_link_libraries(junior_ctrl unitree_sdk2 Threads::Threads)
ament_target_dependencies(junior_ctrl rclcpp)
set_target_properties(junior_ctrl PROPERTIES BUILD_RPATH "${UNITREE_SDK2_ROOT}/lib" INSTALL_RPATH "$ORIGIN/..")
install(TARGETS junior_ctrl DESTINATION lib/${PROJECT_NAME})
install(FILES "${UNITREE_SDK2_ROOT}/lib/libddsc.so" DESTINATION lib RENAME libddsc.so.0)
install(FILES "${UNITREE_SDK2_ROOT}/lib/libddscxx.so" DESTINATION lib RENAME libddscxx.so.0)
ament_package()
